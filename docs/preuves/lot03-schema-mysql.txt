  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `user_id` bigint unsigned NOT NULL,
  `scope` varchar(20) COLLATE utf8mb4_unicode_ci NOT NULL,
  `institution_id` bigint unsigned DEFAULT NULL,
  `academic_year_id` bigint unsigned DEFAULT NULL,
  `role` varchar(40) COLLATE utf8mb4_unicode_ci NOT NULL,
  `status` varchar(20) COLLATE utf8mb4_unicode_ci NOT NULL DEFAULT 'active',
  `local_identifier` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL,
  `starts_at` datetime NOT NULL,
  `ends_at` datetime DEFAULT NULL,
  `suspended_at` datetime DEFAULT NULL,
  `revoked_at` datetime DEFAULT NULL,
  `scope_key` varchar(100) COLLATE utf8mb4_unicode_ci GENERATED ALWAYS AS (concat(`scope`,_utf8mb4':',coalesce(`institution_id`,0),_utf8mb4':',coalesce(`academic_year_id`,0))) STORED,
  `current_slot` tinyint unsigned GENERATED ALWAYS AS ((case when (`status` <> _utf8mb4'closed') then 1 else NULL end)) STORED,
  `current_local_identifier` varchar(100) COLLATE utf8mb4_unicode_ci GENERATED ALWAYS AS ((case when ((`status` <> _utf8mb4'closed') and (`role` = _utf8mb4'learner')) then `local_identifier` else NULL end)) STORED,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `memberships_current_unique` (`user_id`,`scope_key`,`role`,`current_slot`),
  UNIQUE KEY `memberships_local_identifier_unique` (`institution_id`,`academic_year_id`,`current_local_identifier`),
  UNIQUE KEY `memberships_context_unique` (`id`,`user_id`,`institution_id`,`academic_year_id`,`role`),
  KEY `memberships_user_access_index` (`user_id`,`status`,`scope`),
  KEY `memberships_institution_index` (`institution_id`,`academic_year_id`,`status`),
  KEY `memberships_year_fk` (`academic_year_id`,`institution_id`),
  CONSTRAINT `memberships_institution_id_foreign` FOREIGN KEY (`institution_id`) REFERENCES `institutions` (`id`) ON DELETE RESTRICT,
  CONSTRAINT `memberships_user_id_foreign` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE RESTRICT,
  CONSTRAINT `memberships_year_fk` FOREIGN KEY (`academic_year_id`, `institution_id`) REFERENCES `academic_years` (`id`, `institution_id`) ON DELETE RESTRICT,
  CONSTRAINT `memberships_scope_check` CHECK ((((`scope` = _utf8mb4'institution') and (`institution_id` is not null) and (`academic_year_id` is not null) and (`role` in (_utf8mb4'learner',_utf8mb4'teacher',_utf8mb4'institution_admin'))) or ((`scope` = _utf8mb4'platform') and (`institution_id` is null) and (`academic_year_id` is null) and (`role` in (_utf8mb4'catalogue_editor',_utf8mb4'platform_admin')) and (`local_identifier` is null)))),
  CONSTRAINT `memberships_state_check` CHECK (((`status` in (_utf8mb4'active',_utf8mb4'suspended',_utf8mb4'closed')) and ((`ends_at` is null) or (`ends_at` >= `starts_at`)) and ((`status` <> _utf8mb4'suspended') or (`suspended_at` is not null)) and ((`status` <> _utf8mb4'closed') or (`ends_at` is not null) or (`revoked_at` is not null)) and ((`status` <> _utf8mb4'active') or (`revoked_at` is null)) and ((`local_identifier` is null) or (char_length(trim(`local_identifier`)) > 0))))
